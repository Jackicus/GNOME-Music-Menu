// Hook binding to MusicKit JS / Audio Engine export function usePlayerController() { const [isPlaying, setIsPlaying] = useState(false); const [currentTime, setCurrentTime] = useState(0); const [duration, setDuration] = useState(0); const togglePlay = () => { const mk = window.MusicKit?.getInstance(); if (mk?.isPlaying) { mk.pause(); } else { mk.play(); } }; return { isPlaying, currentTime, duration, togglePlay }; }
Linux Desktop Integration (MPRIS2)
  • Connect playback states to D-Bus interface org.mpris.MediaPlayer2.Player.
  • Emit PlaybackStatus signals on play/pause so the GNOME Shell notification panel stays in sync.

Lyrics Synchronization Model
export interface LyricTimestampLine { id: string; startTimeMs: number; endTimeMs: number; text: string; } export function getActiveLyricIndex( lines: LyricTimestampLine[], currentPositionMs: number ): number { return lines.findIndex( line => currentPositionMs >= line.startTimeMs && currentPositionMs < line.endTimeMs ); }
Visual Styling
  • Singing line has text-shadow: 0 0 20px rgba(255,255,255,0.4) and transform: scale(1.04).
  • Non-active lines have reduced opacity (rgba(255,255,255,0.25)).
  • Clicking any lyric invokes musicKit.seekToTime(line.startTimeMs / 1000).
